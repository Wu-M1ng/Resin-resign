package outbound

import (
	"context"
	"fmt"
	"net"
	"strings"
	"testing"
	"time"

	"github.com/Resinat/Resin/internal/subscription"
	"github.com/sagernet/sing/common"
	M "github.com/sagernet/sing/common/metadata"
)

// 诊断用例记录当前缺陷；通过表示成功复现，不表示问题已修复。
func TestDiagnosticHY2Fingerprint(t *testing.T) {
	const supplied = "c083d33c12db8a02d88a5169fb9cee3dae9291fffe38c579891a9dcff55304fb"
	b, err := NewSingboxBuilderWithConfig(SingboxBuilderConfig{
		DNSUpstreams: []string{"udp://127.0.0.1:53", "local"},
	})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = b.Close() })

	// 仅使用本机临时 UDP 端口，不读取或连接真实订阅节点。
	listener, err := net.ListenPacket("udp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = listener.Close() })
	port := listener.LocalAddr().(*net.UDPAddr).Port
	clash := func(extra string) string {
		return fmt.Sprintf(`{"proxies":[{"name":"local-diagnostic","type":"hysteria2","server":"127.0.0.1","port":%d,"password":"fixture-only"%s}]}`, port, extra)
	}
	cases := []struct {
		name, input, buildError, dialError string
	}{
		{"no_fingerprint", clash(""), "", ""},
		{"reported_fingerprint", clash(`,"fingerprint":"` + supplied + `"`), "unknown uTLS fingerprint: " + supplied, ""},
		{"valid_sha256_shape", clash(`,"fingerprint":"` + strings.Repeat("ab", 32) + `"`), "unknown uTLS fingerprint: " + strings.Repeat("ab", 32), ""},
		{"certificate_and_client_fingerprint", clash(`,"fingerprint":"` + supplied + `","client-fingerprint":"chrome"`), "unknown uTLS fingerprint: " + supplied, ""},
		{"client_fingerprint_chrome", clash(`,"client-fingerprint":"chrome"`), "", "unsupported usage for uTLS"},
		{"uri_pin_sha256", fmt.Sprintf("hy2://fixture-only@127.0.0.1:%d?pinSHA256=%s", port, strings.Repeat("ab", 32)), `unknown field "certificate_public_key_sha256"`, ""},
		{"native_public_key_pin", `{"outbounds":[{"type":"hysteria2","tag":"local-public-key-pin","server":"127.0.0.1","domain_resolver":"local","server_ports":["60000:63000"],"password":"fixture-only","tls":{"enabled":true,"insecure":true,"server_name":"fixture.invalid","certificate_public_key_sha256":["AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA="]}}]}`, `unknown field "certificate_public_key_sha256"`, ""},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			nodes, err := subscription.ParseGeneralSubscription([]byte(tc.input))
			if err != nil || len(nodes) != 1 {
				t.Fatalf("parse: nodes=%d err=%v", len(nodes), err)
			}
			t.Logf("converted: %s", nodes[0].RawOptions)
			ob, err := b.Build(nodes[0].RawOptions)
			if tc.buildError != "" {
				if err == nil || !strings.Contains(err.Error(), tc.buildError) {
					_ = common.Close(ob)
					t.Fatalf("build: got %v, want %q", err, tc.buildError)
				}
				t.Logf("build error reproduced: outbound build: %v", err)
				return
			}
			if err != nil {
				t.Fatalf("build: %v", err)
			}
			t.Cleanup(func() { _ = common.Close(ob) })
			t.Log("build succeeded (does not imply remote connectivity)")
			if tc.dialError != "" {
				ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
				defer cancel()
				conn, err := ob.DialContext(ctx, "tcp", M.ParseSocksaddr("127.0.0.1:80"))
				if conn != nil {
					_ = conn.Close()
				}
				if err == nil || !strings.Contains(err.Error(), tc.dialError) {
					t.Fatalf("dial: got %v, want %q", err, tc.dialError)
				}
				t.Logf("loopback dial error reproduced: %v", err)
			}
		})
	}
}
